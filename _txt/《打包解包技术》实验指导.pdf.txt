    // ====== 元数据字段 (二进制存储) ======
    char mode[sizeof(mode_t)];      // 文件类型 + 权限
    char uid[sizeof(uid_t)];        // 属主用户ID

    char mtime[sizeof(time_t)];     // 修改时间
    char atime[sizeof(time_t)];     // 访问时间
    char size[sizeof(off_t)];       // 文件大小 (仅普通文件)
    
    // ====== 设备号 (仅设备文件) ======
    char major[sizeof(unsigned int)];
    char minor[sizeof(unsigned int)];

    char filePathLen[sizeof(int)];  // 文件名实际长度

    // ====== 类型标识 ======
    char type;  // '-' 普通文件 | 'D' 目录 | 'L' 软链接 | 'B' 块设备

3.2 lstat()  与 stat()  的关键区别
 
代码在 traverseDir  的过滤中使用了 lstat() ，这是正确且至关重要的。
错误示例：如果使用 stat() ，遇到指向目录的软链接，会返回 S_ISDIR  为真，导致遍历器误入并死循环。
3.3 路径处理技巧：chdir  的使用
 
好处：
1. 所有路径变为相对路径，打包时记录更短。
2. 解包时灵活性更高（可解压到任意目标目录）。
3. 避免绝对路径带来的安全问题。

int CMyPack::packHeaderRecord(Utils::CSimpleFile &sf, 
                               const std::string &file, 
                               struct stat &buff) {
    recordStruct record;
    memset(&record, 0, sizeof(record));
    
    // 1. 拷贝元数据
    memcpy(record.mode, &buff.st_mode, sizeof(buff.st_mode));
    memcpy(record.uid, &buff.st_uid, sizeof(buff.st_uid));
    memcpy(record.gid, &buff.st_gid, sizeof(buff.st_gid));
    memcpy(record.mtime, &buff.st_mtime, sizeof(buff.st_mtime));
    memcpy(record.atime, &buff.st_atime, sizeof(buff.st_atime));
    
    // 2. 根据文件类型填充
    if (S_ISREG(buff.st_mode)) {
        record.type = '-';
        memcpy(record.size, &buff.st_size, sizeof(buff.st_size));
    } else if (S_ISDIR(buff.st_mode)) {
        record.type = 'D';
    } else if (S_ISLNK(buff.st_mode)) {
        record.type = 'L';
        readlink(file.c_str(), linkPath, sizeof(linkPath));
        memcpy(record.linkPath, linkPath, ...);
    } else if (S_ISBLK(buff.st_mode)) {
        record.type = 'B';
        // 记录设备号
        unsigned int major = major(buff.st_dev);
        unsigned int minor = minor(buff.st_dev);
        memcpy(record.major, &major, sizeof(major));
        memcpy(record.minor, &minor, sizeof(minor));
    }

4.3 数据块写入流程
 
关键点：
每次读取最多 512 字节，写入时固定 512 字节。
最后一次读取不足 512 字节时，由于 memset  已将 data.block  清零，写入的块中有效数据后的部分自
动补零。
这保证了数据块严格 512 字节对齐，便于 lseek  跳过。

int CMyPack::packDataRecord(Utils::CSimpleFile &sf, 
                             const std::string &file, 
                             struct stat &buff) {
    recordStruct data;
    memset(data.block, 0, sizeof(data.block));
    
    CSimpleFile regularFile(file, O_RDONLY, -1);
    
    int ret = 0;
    while ((ret = regularFile.readN(data.block, sizeof(data.block))) > 0) {
        // 每次写入 512 字节块 (不足的已用 memset 补零)
        sf.writeN(data.block, sizeof(data.block));
        memset(data.block, 0, sizeof(data.block));
    }
    return 0;
}

5.2 数据块读取逻辑
 
关键点：
数据块在打包时按 512 字节写入（含填充），解包时根据 size  只读取有效部分。
最后一块的填充数据被丢弃，恢复的文件大小精确为 size_ 。

9. 课堂思考题
 
Q1：遍历时 readdir  的顺序不确定，会影响打包/解包的正确性吗？
 
答案：不影响，但有前提条件——递归遍历时必须在进入子目录之前先记录目录自身（先序遍历）。
同层文件的顺序不影响正确性（它们互不依赖）。
父目录必须排在子文件之前（否则解包时目录不存在，open  子文件失败）。
代码中的 traverseDir  在递归调用 traverseDir(subdir)  之前先执行了 
result.push_back(path) ，保证了这一点。
Q2：如果源目录中包含指向自身的软链接（ln -s . self ），遍历会死循
环吗？
 
答案：不会。因为 entry->d_type  对软链接返回 DT_LNK ，而不是 DT_DIR 。代码中只有 entry->d_type == 
DT_DIR  时才递归。即使使用 lstat  判断，S_ISLNK  也为真，不会进入递归分支。
但如果误用 stat() ：stat("self")  会返回目标目录的信息，S_ISDIR  为真，导致递归进入，造成死循环。
这就是必须使用 lstat()  的原因！

Q3：为什么解包时要强制目标目录为空？
 
答案：防止覆盖用户现有数据。unpackCheck  中会遍历目标目录，如果发现任何非 .  和 ..  的条目，直接报错
返回 DESTINATION_DIR_NOT_EMPTY 。
Q4：打包后的文件，数据块为什么要固定 512 字节？为什么不用文件实际大
小？
 
答案：
1. 对齐：便于使用 lseek  快速跳转到任意文件的数据区。
2. 简化读写：每次读写固定大小，代码简单，减少碎片。
3. 历史原因：Tar 格式使用 512 字节块（磁盘扇区大小）。
4. 代价：小文件会浪费空间（如 1 字节的文件占用 512 字节存储）。
